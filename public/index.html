            Client-Side DOM Helper
            
5-Second Browser Console Extractor for JS-Rendered Directories

            

              For client-side rendered tables like https://high-paying-companies.netlify.app/, run this snippet in DevTools Console (F12) to copy all 212 companies and pipe them into POST /api/workflow/trigger.
            

          

          
        

        
// Run in Chrome/Edge/Firefox DevTools Console (F12) on https://high-paying-companies.netlify.app/
copy(
  JSON.stringify(
    Array.from(document.querySelectorAll('table tbody tr')).map((tr, idx) => {
      const cells = tr.querySelectorAll('td');
      const originalUrl = tr.querySelector('a')?.href || '';
      return {
        name: cells[0]?.innerText.trim() || `Company-${idx + 1}`,
        ctcBand: cells[1]?.innerText.trim() || 'High Tier',
        workMode: cells[2]?.innerText.trim() || 'Hybrid',
        originalUrl,
        batchNumber: Math.floor(idx / 20) + 1
      };
    }),
    null,
    2
  )
);
console.log('Copied all 212 companies as JSON payload for Cloudflare Workflow!');
